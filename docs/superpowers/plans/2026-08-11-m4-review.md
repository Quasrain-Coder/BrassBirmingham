```ts
export interface AnnotatedStep {
  seq: number;        // 落库行动序号
  round: number;      // 行动发生时的轮次
  era: Era;
  action: Action;
  actionSummary: string;   // describeAction(行动前状态, seat, action)
  actualScore: number;
  bestScore: number;
  gap: number;             // bestScore - actualScore（≥0）
  bestAction: Action;
  bestSummary: string;     // describeAction(行动前状态, seat, bestAction)
}
export interface EraStanding { vp: number; incomeLevel: number; rank: number }
export interface TopPoint extends AnnotatedStep { situation: string }  // summarizeState(行动前状态, seat)
export interface GameAnnotation {
  seat: PlayerIndex;
  steps: AnnotatedStep[];      // 该座位全部决策点，seq 升序（曲线数据）
  topPoints: TopPoint[];       // gap 降序 topN（并列取 seq 小者），附局面摘要
  canalEnd: EraStanding;       // 运河时代清算完成后的站位
  gameEnd: EraStanding;        // 终局站位
}
export const REVIEW_TOP_N = 10;
export function annotateGame(
  playerCount: 2 | 3 | 4,
  seed: number,
  log: { seq: number; player: number; action: Action }[],
  seat: PlayerIndex,
  topN?: number,   // 缺省 REVIEW_TOP_N
): GameAnnotation;
```

**实现要点（两遍重放）：**
- Pass 1：`newGame(playerCount, seed)` 起手；逐条 log：`entry.player === seat` 时先标注再 `applyAction`——`legal = enumerateActions(state, seat)`；逐个 `scoreAction` 求 best（**并列取数组序靠前者**，与 prescreen 口径一致）；`actualScore = scoreAction(state, seat, entry.action)`；`gap = Math.max(0, bestScore - actualScore)`（浮点防御）。每条 log 应用后检测 `state.era` 跳变（canal→rail）→ 记录 canalEnd；终局记录 gameEnd。`rank` = 全体玩家 vp 降序排名（并列 vp 按座位号小者居前，与 engine 终局平局链无关、仅展示用）。
- Pass 2：仅当 topPoints 非空——按 pass 1 的 gap 降序（并列 seq 升序）取 topN 得 seq 集合，重新从 `newGame` 重放，在命中 seq 处 capture `summarizeState(state, seat)` 填入 `situation`。
- 该座位 0 决策点（理论边界）：steps/topPoints 为空数组，canalEnd/gameEnd 仍返回（task 3 据此走模板兜底）。

```ts
/**
 * 重放标注器（确定性，零 token）——M4 复盘的客观锚点。
 *
 * 重放落库 action log，在被标注座位的每个决策点重新 enumerateActions 并用
 * scoreAction 给全部候选打分：产出每步分差（bestScore - actualScore）、
 * 分差 Top N 决策点（附 summarizeState 局面摘要）、运河时代结束与终局站位。
 * 两遍重放：pass 1 算分差序列，pass 2 仅为 topN seq 补局面摘要（summarizeState
 * 较贵，不为每步都算）。同 seed 同 log 输出逐字节一致（测试锁定）。
 */
import {
  applyAction,
  enumerateActions,
  incomeLevelAt,
  newGame,
} from '@brass/engine';
import type { Action, Era, GameState, PlayerIndex } from '@brass/engine';
import { scoreAction } from './heuristic.js';
import { describeAction, summarizeState } from './summarize.js';

export const REVIEW_TOP_N = 10;

export interface AnnotatedStep { /* …按 Interfaces 块逐字段定义… */ }
export interface EraStanding { vp: number; incomeLevel: number; rank: number }
export interface TopPoint extends AnnotatedStep { situation: string }
export interface GameAnnotation { /* …按 Interfaces 块逐字段定义… */ }

function standing(state: GameState, seat: PlayerIndex): EraStanding {
  const order = state.players
    .map((p, i) => ({ vp: p.vp, seat: i }))
    .sort((a, b) => b.vp - a.vp || a.seat - b.seat);
  const rank = order.findIndex((o) => o.seat === seat) + 1;
  const ps = state.players[seat]!;
  return { vp: ps.vp, incomeLevel: incomeLevelAt(ps.incomeSpace), rank };
}

export function annotateGame(
  playerCount: 2 | 3 | 4,
  seed: number,
  log: { seq: number; player: number; action: Action }[],
  seat: PlayerIndex,
  topN: number = REVIEW_TOP_N,
): GameAnnotation {
  // ---- pass 1：分差序列 + 时代站位 ----
  let state = newGame(playerCount, seed);
  const steps: AnnotatedStep[] = [];
  let canalEnd: EraStanding | null = null;
  for (const entry of log) {
    const prevEra = state.era;
    if (entry.player === seat) {
      const legal = enumerateActions(state, seat);
      let best = legal[0]!;
      let bestScore = Number.NEGATIVE_INFINITY;
      for (const candidate of legal) {
        const s = scoreAction(state, seat, candidate);
        if (s > bestScore) { bestScore = s; best = candidate; }  // 并列保留数组序靠前者
      }
      const actualScore = scoreAction(state, seat, entry.action);
      steps.push({
        seq: entry.seq,
        round: state.round,
        era: state.era,
        action: entry.action,
        actionSummary: describeAction(state, seat, entry.action),
        actualScore,
        bestScore,
        gap: Math.max(0, bestScore - actualScore),
        bestAction: best,
        bestSummary: describeAction(state, seat, best),
      });
    }
    state = applyAction(state, entry.action);
    if (prevEra === 'canal' && state.era === 'rail') canalEnd = standing(state, seat);
  }
  const gameEnd = standing(state, seat);
  // 边界：对局在运河时代内终止（不会有）时 canalEnd 兜底为 gameEnd
  const canal = canalEnd ?? gameEnd;

  // ---- pass 2：topN 局面摘要 ----
  const top = [...steps]
    .sort((a, b) => b.gap - a.gap || a.seq - b.seq)
    .slice(0, Math.max(0, topN));
  const wanted = new Set(top.map((t) => t.seq));
  const situations = new Map<number, string>();
  if (wanted.size > 0) {
    let s2 = newGame(playerCount, seed);
    for (const entry of log) {
      if (wanted.has(entry.seq)) situations.set(entry.seq, summarizeState(s2, seat));
      s2 = applyAction(s2, entry.action);
    }
  }
  const topPoints = top.map((t) => ({ ...t, situation: situations.get(t.seq) ?? '' }));
  return { seat, steps, topPoints, canalEnd: canal, gameEnd };
}
```

```ts
export type CompleteStep =
  | CompleteResponse
  | Error
  | ((req: CompleteRequest) => CompleteResponse);

export function makeCompleteResponse(
  text: string,
  usage: { input: number; output: number } = { input: 10, output: 5 },
): CompleteResponse {
  return { text, usage };
}

// FixtureClient 内新增：
readonly completeRequests: CompleteRequest[] = [];
defaultCompleteResponse: CompleteResponse = makeCompleteResponse('fixture text');
private readonly completeScript: CompleteStep[];

// 构造函数改为 constructor(script: FixtureStep[] = [], opts?: { completeScript?: CompleteStep[] })
// ——原签名是 ...script: FixtureStep[]，改成数组首参会破坏 llm-agent.test.ts 现有调用，
// 故保留 ...script 不变，completeScript 改**可写属性**：
//   readonly 改 public completeScriptQueue: CompleteStep[] = []，测试直接 push。
// 二选一实现，以不破坏现有 decide 调用方为准（llm-agent.test.ts 用了 ...script 变参）。

complete(req: CompleteRequest): Promise<CompleteResponse> {
  this.completeRequests.push(req);
  const step = this.completeScriptQueue.shift();
  if (step === undefined) return Promise.resolve(this.defaultCompleteResponse);
  if (step instanceof Error) return Promise.reject(step);
  if (typeof step === 'function') return Promise.resolve(step(req));
  return Promise.resolve(step);
}
```

```ts
  /** 降级原因汇总（写报告元信息；无降级为空串）。 */
  note: string;
}
export class ReviewAgent {
  constructor(client: ClaudeClient, opts?: { model?: string; limits?: Partial<ReviewLimits> });
  analyze(annotation: GameAnnotation): Promise<ReviewAnalysis>;
}
```

**实现要点：**
- model 解析：`opts?.model ?? process.env['BRASS_REVIEW_MODEL'] ?? 'claude-opus-5'`（构造时一次解析存字段——避免 client.ts:78 那种 env 死代码）。
- `analyze` 流程：
  1. `annotation.topPoints.length === 0` → 立即返回 `{ analysis: '', model, usage: {0,0}, degraded: true, note: '无决策点可分析' }`。
  2. topPoints 按 batchSize 分批；逐批 `complete(batchPrompt)`。**限额**：第 2 批起，若累计 `usage.input >= maxInputTokens * 0.6` 则跳过剩余批次（note 记"限额跳批 N 点"），被跳点仅以数据行进入综合 prompt。
  3. 批次失败（异常或返回空 text）→ catch 记录 note，继续下一批；该批点按跳批同待遇。
  4. 全部批次失败/跳批且无已分析片段 → 不调用综合，返回 `analysis: ''`、degraded、note。
  5. 有片段 → 综合调用 `complete(synthesisPrompt)`；失败/空 text → degraded，`analysis = 确定性拼接(片段)`（见下）。
- **批次 prompt**（`buildBatchPrompt(annotation, chunk, batchIdx, batchCount)`）：system 静态（角色：Brass: Birmingham 复盘教练；任务：逐点判断启发式分差是否真错误——识别长线布局并辩护，否则给具体替代操作与原因链；引用具体轮次与板块，禁止空泛建议；输出 markdown，每点一个 `### 第N手（第R轮·时代）` 小节）。user = 每点依次拼：`situation` + `实际选择：actionSummary（得分 actualScore）` + `启发式最优：bestSummary（得分 bestScore，分差 gap）`。
- **综合 prompt**（`buildSynthesisPrompt(annotation, fragments, skippedLines)`）：user = 运河/终局站位数据行 + 各批片段 + 被跳点数据行；要求按固定结构成稿：`## 总评`（打法风格识别）→ `## 关键转折点`（从片段提炼 3–5 手）→ `## 时代切换站位` → `## 下局建议`（1–3 条可执行）。
- 确定性拼接降级：`analysis = '## 关键决策点逐手分析（LLM 综合不可用）\n\n' + fragments.join('\n\n')`。
- usage 为各次调用之和；任何降级路径 note 非空。

```ts
export class ReviewAgent {
  private readonly client: ClaudeClient;
  private readonly model: string;
  private readonly limits: ReviewLimits;

  constructor(client: ClaudeClient, opts?: { model?: string; limits?: Partial<ReviewLimits> }) {
    this.client = client;
    this.model = opts?.model ?? process.env['BRASS_REVIEW_MODEL'] ?? 'claude-opus-5';
    this.limits = { ...REVIEW_LIMITS, ...opts?.limits };
  }

  async analyze(annotation: GameAnnotation): Promise<ReviewAnalysis> {
    const usage = { input: 0, output: 0 };
    const notes: string[] = [];
    if (annotation.topPoints.length === 0) {
      return { analysis: '', model: this.model, usage, degraded: true, note: '无决策点可分析' };
    }
    const chunks: TopPoint[][] = [];
    for (let i = 0; i < annotation.topPoints.length; i += this.limits.batchSize) {
      chunks.push(annotation.topPoints.slice(i, i + this.limits.batchSize));
    }
    const fragments: string[] = [];
    const skipped: TopPoint[] = [];
    for (const [idx, chunk] of chunks.entries()) {
      if (idx > 0 && usage.input >= this.limits.maxInputTokens * 0.6) {
        skipped.push(...chunk);
        notes.push(`限额跳批 ${chunk.length} 点（累计 input ${usage.input}）`);
        continue;
      }
      try {
        const { system, user } = buildBatchPrompt(annotation, chunk, idx, chunks.length);
        const res = await this.client.complete({
          system, user, model: this.model,
          maxTokens: this.limits.batchMaxTokens, timeoutMs: this.limits.timeoutMs,
        });
        usage.input += res.usage.input;
        usage.output += res.usage.output;
        if (res.text.trim() === '') throw new Error('空响应');
        fragments.push(res.text);
      } catch (err) {
        skipped.push(...chunk);
        notes.push(`批次 ${idx + 1} 失败：${err instanceof Error ? err.message : String(err)}`);
      }
    }
    if (fragments.length === 0) {
      return { analysis: '', model: this.model, usage, degraded: true, note: notes.join('；') };
    }
    try {
      const { system, user } = buildSynthesisPrompt(annotation, fragments, skipped);
      const res = await this.client.complete({
        system, user, model: this.model,
        maxTokens: this.limits.synthesisMaxTokens, timeoutMs: this.limits.timeoutMs,
      });
      usage.input += res.usage.input;
      usage.output += res.usage.output;
      if (res.text.trim() === '') throw new Error('空响应');
      return { analysis: res.text, model: this.model, usage, degraded: notes.length > 0, note: notes.join('；') };
    } catch (err) {
      notes.push(`综合失败：${err instanceof Error ? err.message : String(err)}`);
      const analysis = `## 关键决策点逐手分析（LLM 综合不可用）\n\n${fragments.join('\n\n')}`;
      return { analysis, model: this.model, usage, degraded: true, note: notes.join('；') };
    }
  }
}
```

`buildBatchPrompt`/`buildSynthesisPrompt` 为模块内私有函数；system 静态常量（不含任何对局数据，缓存友好）。


**server 实现要点（ws.ts）：**
- `GameServerOptions` 加注入缝（沿用 aiAgentFactory 模式）：

```ts
/** 复盘分析器注入缝：缺省 new ReviewAgent(new AnthropicClient())。测试用 stub。 */
reviewAnalyzerFactory?: () => Pick<ReviewAgent, 'analyze'>;
```

- `handleRequestReview(conn, msg)`：
  1. 校验：`gameId` 非空 string、`seat` 为整数；`getGame` 未命中 → `sendError(conn, 'unknown-game', ...)`；`status !== 'finished'` → `sendError(conn, 'game-not-finished', ...)`；`seat >= playerCount` → `sendError(conn, 'invalid-seat', ...)`。
  2. 缓存：`!msg.regenerate` 且 `getReview` 命中 → 直接 `send(conn, { type: 'review_result', report: storedToReport(row) })`。
  3. 在飞守卫：`const inflight = new Map<string, Promise<void>>()`（key `${gameId}:${seat}`）；已存在 → 回 `review_status generating` 后 return（不重复烧 token）。
  4. 生成：`send(conn, review_status generating)` → async 管线：`listActions` → `annotateGame(game.playerCount, game.seed, log, seat)` → `analyzer.analyze(annotation)` → `assembleReport(...)` + `buildCurve(...)` → `upsertReview` → `send(conn, review_result)`。异常 → `send(conn, review_status failed + message)` + `console.error`。`finally` 清 inflight。**注意 conn 可能已断开——send 已有 readyState 守卫，无需额外处理；落库照常完成（别的请求者能拿缓存）。**
  5. `storedToReport(row)`：`JSON.parse(row.curve)` 还原 CurvePoint[]；`usage = { input: row.inputTokens, output: row.outputTokens }`。
  6. llm 的 `CurvePoint` 与 protocol 的 `CurvePoint` 结构一致（都引用 engine `Era`），直接赋值靠结构类型兼容，**加一行注释说明**。
- `broadcastGameOver`：`game_over` 消息加 `gameId: entry.session.gameId`。
- `routeMessage` switch 加 `case 'request_review'`。
- main.ts：默认 factory 接 `AnthropicClient`（M3 已有同款接线可抄）。
